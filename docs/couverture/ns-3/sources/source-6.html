     public static String convertirHeure(String heureEnSecondes) {
         try {
             double totalSecondes = Double.parseDouble(heureEnSecondes);
 
             int h = (int) (totalSecondes / 3600);
             int m = (int) ((totalSecondes % 3600) / 60);
 
             h = h % 24;
